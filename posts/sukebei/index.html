    go代码 # 

    package main

import (
	"io"
	"log"
	"net/http"
	"net/url"
	"regexp"
	"strings"
	"time"
)

const (
	HttpProxy  = "http://10.0.0.211:20172"
	ListenAddr = "0.0.0.0:8080"
	UserAgent  = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/150.0.0.0 Safari/537.36"
)

func main() {
	log.Println("[INIT] 正在初始化 Go Proxy 服务...")

	proxyUrl, err := url.Parse(HttpProxy)
	if err != nil {
		log.Fatalf("[FATAL] 代理 URL 解析失败 (%s): %v", HttpProxy, err)
	}
	log.Printf("[INIT] 成功配置上游 HTTP 代理: %s", HttpProxy)

	transport := &http.Transport{
		Proxy: http.ProxyURL(proxyUrl),
	}
	client := &http.Client{
		Transport: transport,
		Timeout:   30 * time.Second,
	}

	http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
		handleGoProxy(w, r, client)
	})

	log.Printf("[READY] Go 降级服务已就绪，正在监听: %s", ListenAddr)
	if err := http.ListenAndServe(ListenAddr, nil); err != nil {
		log.Fatalf("[FATAL] 服务异常退出: %v", err)
	}
}

func handleGoProxy(w http.ResponseWriter, r *http.Request, client *http.Client) {
	targetURL := "https:/" + r.URL.Path
	//if strings.Contains(targetURL, ".skt") || strings.Contains(targetURL, ".js") || strings.Contains(targetURL, ".css") {
	//	w.WriteHeader(http.StatusNotFound)
	//	return
	//}
	if strings.Contains(r.URL.Path, "imagetwist") {

		targetURL = "https:/" + r.URL.Path

		// 请求页面获取真实地址
		req, _ := http.NewRequest("GET", targetURL, nil)
		req.Header.Set("User-Agent", UserAgent)
		resp, _ := client.Do(req)
		defer resp.Body.Close()

		body, _ := io.ReadAll(resp.Body)
		content := string(body)

		// 打印页面内容
		log.Println("页面内容:", content)

		// 提取并打印所有图片地址
		reImg := regexp.MustCompile(`src=["']([^"']+)["']`)
		imgMatches := reImg.FindAllStringSubmatch(content, -1)
		for _, match := range imgMatches {
			if strings.Contains(match[1], "imagetwist.com/i/") {
				log.Println("图片:", match[1])
				targetURL = match[1]
				break
			}
		}
	}
	if strings.Contains(r.URL.Path, "/upload/en/") {
		r.URL.Path = strings.Replace(r.URL.Path, "/upload/en/", "/upload/", 1)
		targetURL = "https:/" + r.URL.Path

		// 请求页面获取真实地址
		req, _ := http.NewRequest("GET", targetURL, nil)
		req.Header.Set("User-Agent", UserAgent)
		resp, _ := client.Do(req)
		defer resp.Body.Close()

		body, _ := io.ReadAll(resp.Body)
		content := string(body)

		// 打印页面内容
		//log.Println("页面内容:", content)

		// 提取并打印所有图片地址
		reImg := regexp.MustCompile(`src=["']([^"']+)["']`)
		imgMatches := reImg.FindAllStringSubmatch(content, -1)
		for _, match := range imgMatches {
			if strings.Contains(match[1], "Application") {
				log.Println("图片:", match[1])
				targetURL = match[1]
				break
			}
		}
	}

	req, _ := http.NewRequest(r.Method, targetURL, r.Body)

	resp, err := client.Do(req)
	if resp != nil {
		defer resp.Body.Close()
	}
	if err != nil {
		http.Error(w, "代理请求失败: "+err.Error(), http.StatusBadGateway)
		return
	}

	for k, v := range resp.Header {
		w.Header()[k] = v
	}
	w.WriteHeader(resp.StatusCode)

	_, _ = io.Copy(w, resp.Body)
}

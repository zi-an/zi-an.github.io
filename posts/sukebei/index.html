    go代码 # 

    package main

import (
	"fmt"
	"io"
	"log"
	"net/http"
	"net/url"
	"strings"
	"time"
)

const (
	HTTP_PROXY  = "http://10.0.0.211:20172" // HTTP 代理地址
	LISTEN_ADDR = "127.0.0.1:8080"         // Go 中转服务监听地址
)

func main() {
	log.Println("[INIT] 正在初始化 Go Fallback Proxy 服务...")

	// 1. 解析 HTTP 代理 URL
	proxyURL, err := url.Parse(HTTP_PROXY)
	if err != nil {
		log.Fatalf("[FATAL] 代理 URL 解析失败 (%s): %v", HTTP_PROXY, err)
	}
	log.Printf("[INIT] 成功配置上游 HTTP 代理: %s", HTTP_PROXY)

	// 2. 创建带代理的 HTTP Transport & Client
	transport := &http.Transport{
		Proxy: http.ProxyURL(proxyURL),
	}
	client := &http.Client{
		Transport: transport,
		Timeout:   30 * time.Second,
	}

	// 3. 注册处理路由
	http.HandleFunc("/fallback", func(w http.ResponseWriter, r *http.Request) {
		reqStart := time.Now()
		log.Println("----------------------------------------------------------------")
		log.Printf("[REQ START] 收到来自 Nginx 的降级请求: %s %s", r.Method, r.URL.String())

		// 读取 Nginx 传过来的 Header 参数
		targetHost := r.Header.Get("X-Target-Host")
		targetURI := r.Header.Get("X-Target-URI")

		log.Printf("[PARAM] 读取请求头 X-Target-Host = '%s'", targetHost)
		log.Printf("[PARAM] 读取请求头 X-Target-URI  = '%s'", targetURI)

		if targetHost == "" {
			log.Println("[ERROR] 缺失必要参数 X-Target-Host，直接返回 400")
			http.Error(w, "Missing X-Target-Host header", http.StatusBadRequest)
			return
		}

		// ------------------ 拦截逻辑 ------------------
		// 1. 域名拦截（保留原有的 .skt 检查）
		if strings.Contains(strings.ToLower(targetHost), ".skt") {
			log.Printf("[BLOCK] 监测到无效/伪域名 (%s)，直接拒绝连接，返回 404", targetHost)
			w.WriteHeader(http.StatusNotFound)
			w.Write([]byte("404 Not Found (Blocked Domain)"))
			log.Printf("[REQ END] 请求被拦截并返回 404，耗时: %v", time.Since(reqStart))
			log.Println("----------------------------------------------------------------")
			return
		}

		// 2. 非图片请求拦截
		pathLower := strings.ToLower(targetURI)
		allowedExts := []string{".jpg", ".jpeg", ".png", ".gif", ".webp", ".ico", ".bmp", ".svg", ".avif"}

		isImage := false
		for _, ext := range allowedExts {
			// 支持带参数的图片请求（例如 /image.jpg?v=1.0）
			if strings.HasSuffix(pathLower, ext) || strings.Contains(pathLower, ext+"?") {
				isImage = true
				break
			}
		}

		if !isImage {
			log.Printf("[BLOCK] 监测到非图片请求 (%s)，直接返回 404", targetURI)
			w.WriteHeader(http.StatusNotFound)
			w.Write([]byte("404 Not Found (Only images allowed)"))
			log.Printf("[REQ END] 请求被拦截并返回 404，耗时: %v", time.Since(reqStart))
			log.Println("----------------------------------------------------------------")
			return
		}
		// -------------------------------------------------------

		// 拼接目标 URL
		targetURL := fmt.Sprintf("https://%s/%s", targetHost, targetURI)
		if r.URL.RawQuery != "" {
			targetURL += "?" + r.URL.RawQuery
		}
		log.Printf("[TARGET] 准备通过代理请求真实目标: %s", targetURL)

		// 构造新的 HTTP 请求
		req, err := http.NewRequest(r.Method, targetURL, r.Body)
		if err != nil {
			log.Printf("[ERROR] 创建出站请求失败: %v", err)
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}

		// 复制 Header 并重置 Host/Referer
		log.Println("[HEADER] 正在透传并修正 Header...")
		for k, v := range r.Header {
			if k != "X-Target-Host" && k != "X-Target-URI" && k != "Host" {
				req.Header[k] = v
			}
		}
		req.Header.Set("Host", targetHost)
		req.Header.Set("Referer", fmt.Sprintf("https://%s", targetHost))

		// 开始发起请求
		log.Printf("[FETCH] 开始发送代理请求 (超时时间: 30s)...")
		fetchStart := time.Now()
		resp, err := client.Do(req)
		fetchCost := time.Since(fetchStart)

		if err != nil {
			log.Printf("[FETCH FAIL] 通过代理访问目标失败! 耗时: %v, 错误: %v", fetchCost, err)
			http.Error(w, fmt.Sprintf("Proxy fetch failed: %v", err), http.StatusBadGateway)
			return
		}
		defer resp.Body.Close()

		log.Printf("[FETCH SUCCESS] 成功拿到响应! 目标返回状态码: %d, 耗时: %v", resp.StatusCode, fetchCost)

		// 复制响应头回传给 Nginx
		for k, v := range resp.Header {
			w.Header()[k] = v
		}
		w.Header().Set("X-Proxy-Via", "Go-Fallback-Proxy")
		w.WriteHeader(resp.StatusCode)

		// 复制响应体
		log.Println("[STREAM] 开始向 Nginx 写回响应 Body...")
		written, err := io.Copy(w, resp.Body)
		if err != nil {
			log.Printf("[ERROR] 回写响应 Body 过程中断: %v", err)
		} else {
			log.Printf("[STREAM DONE] 响应 Body 写入完毕，传输字节数: %d bytes", written)
		}

		log.Printf("[REQ END] 本次降级请求处理完成，总耗时: %v", time.Since(reqStart))
		log.Println("----------------------------------------------------------------")
	})

	log.Printf("[READY] Go 降级服务已就绪，正在监听: %s", LISTEN_ADDR)
	if err := http.ListenAndServe(LISTEN_ADDR, nil); err != nil {
		log.Fatalf("[FATAL] 服务异常退出: %v", err)
	}
}
